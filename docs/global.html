(async) WDataScheduler(optopt) → {Object}

Description:

  • 基於檔案之數據變更驅動任務建構器

    每輪依序取得最新數據與既有數據、比對差異，再逐項呼叫funRemove、funAdd、funModify。逐項處理中任一項失敗（函數同步拋錯、回傳之Promise被reject，或函數成功而其後任務狀態更新失敗）不中止本輪：其餘項照常處理，該項記錄error事件並列入本輪結果之failed，逐項處理結束後另發出1筆warn事件fail-items-stage-main。失敗項之任務狀態不更新，只要funGetCurrent未將其視為已完成，下輪仍比對出差異而重試，故funRemove、funAdd、funModify須可重複執行。取清單與比對階段（funAfterStart、funGetNew、funGetCurrent、比對）之錯誤則取消本輪，不進入逐項處理

    本輪結果物件之格式為{ success: { add, modify, remove }, failed, msgCancel }：success之各陣列為處理成功之主鍵值；failed為失敗項陣列，各項為{ id, mode, event, msg }，id為主鍵值，mode為'add'、'modify'或'remove'，event為該階段之事件名，msg為固定詞彙（'funAdd failed'、'funModify failed'、'funRemove failed'或'failed to update task state'），原錯誤訊息只記於該項之error事件；msgCancel為本輪未進入逐項處理時之取消訊息（階段層級錯誤、'no data'、'no difference'等），否則為空字串。判定某項是否完成應以success為準，未列於success者一律視為未完成。funBeforeEnd與end事件之監聽函數各收到一份獨立之結果物件

Example
import w from 'wsemi'
import WDataScheduler from './src/WDataScheduler.mjs'

//fdTagRemove
let fdTagRemove = './_tagRemove'
w.fsCleanFolder(fdTagRemove)

//fdTaskCpActualSrc
let fdTaskCpActualSrc = './_taskCpActualSrc'
w.fsCleanFolder(fdTaskCpActualSrc)

//fdTaskCpSrc
let fdTaskCpSrc = './_taskCpSrc'
w.fsCleanFolder(fdTaskCpSrc)

//funGetNew
let funGetNew = async() => {
    let items = [
        {
            'id': '114115',
            'hash': 'abc',
        },
    ]
    return items
}

//funGetCurrent
let funGetCurrent = async() => {
    let items = [
        {
            'id': '114115',
            'hash': 'abc',
        },
        {
            'id': '114116',
            'hash': 'def',
        },
    ]
    return items
}

//funRemove
let funRemove = async(v) => {
    //do somethings
}

//funAdd
let funAdd = async(v) => {
    //do somethings
}

//funModify
let funModify = async(v) => {
    //do somethings
}

let opt = {
    fdTagRemove,
    fdTaskCpActualSrc,
    fdTaskCpSrc,
    funGetNew,
    funGetCurrent,
    funRemove,
    funAdd,
    funModify,
}
let ev = await WDataScheduler(opt)
    .catch((err) => {
        console.log(err)
    })
ev.on('change', (msg) => {
    delete msg.type
    delete msg.timeRunStart
    delete msg.timeRunEnd
    delete msg.timeRunSpent
    console.log('change', msg)
})
// change { event: 'start', msg: 'running...' }
// change { event: 'proc-callfun-getNew', msg: 'start...' }
// change { event: 'proc-callfun-getNew', num: 2, msg: 'done' }
// change { event: 'proc-callfun-getCurrent', msg: 'start...' }
// change { event: 'proc-callfun-getCurrent', num: 1, msg: 'done' }
// change { event: 'proc-compare', msg: 'start...' }
// change {
//   event: 'proc-compare',
//   numRemove: 0,
//   numAdd: 1,
//   numModify: 0,
//   numSame: 1,
//   msg: 'done'
// }
// change { event: 'proc-add-callfun-add', id: '114116', msg: 'start...' }
// change { event: 'proc-add-callfun-add', id: '114116', msg: 'done' }
// change { event: 'end', msg: 'done' }
